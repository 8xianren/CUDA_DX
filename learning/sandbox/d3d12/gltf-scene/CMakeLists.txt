cmake_minimum_required(VERSION 3.24)
project(scene_import_probe LANGUAGES C CXX)
include(FetchContent)
# 固定源码提交，避免每次构建悄悄更换解析行为；依赖与产物均留在本课 build 下。
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_ASSIMP_TOOLS OFF CACHE BOOL "" FORCE)
set(ASSIMP_INSTALL OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_ZLIB ON CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_ALL_IMPORTERS_BY_DEFAULT OFF CACHE BOOL "" FORCE)
foreach(format GLTF FBX OBJ COLLADA PLY STL)
  set(ASSIMP_BUILD_${format}_IMPORTER ON CACHE BOOL "" FORCE)
endforeach()
set(ASSIMP_BUILD_ALL_EXPORTERS_BY_DEFAULT OFF CACHE BOOL "" FORCE)
# 导出器仅用于生成等价 GLB 验证资产，不自行编写 GLB 打包器。
set(ASSIMP_BUILD_GLTF_EXPORTER ON CACHE BOOL "" FORCE)
FetchContent_Declare(assimp
  URL https://codeload.github.com/assimp/assimp/zip/fb375dd8c0a032106a2122815fb18dffe0283721
  URL_HASH SHA256=e4391db53f34cfc5c979df9b13b05c37d1b47d0ab516a32df6670e6939eb71d1)
FetchContent_MakeAvailable(assimp)
add_executable(scene_import_probe src/import_probe.cpp)
target_compile_features(scene_import_probe PRIVATE cxx_std_17)
target_link_libraries(scene_import_probe PRIVATE assimp)
target_compile_options(scene_import_probe PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
add_executable(mesh_layout_test src/cpu_mesh.cpp src/mesh_layout_test.cpp)
target_compile_features(mesh_layout_test PRIVATE cxx_std_17)
target_link_libraries(mesh_layout_test PRIVATE assimp)
target_compile_options(mesh_layout_test PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
add_executable(scene_hierarchy_test src/cpu_mesh.cpp src/cpu_material.cpp src/cpu_scene.cpp src/scene_hierarchy_test.cpp)
target_compile_features(scene_hierarchy_test PRIVATE cxx_std_17)
target_link_libraries(scene_hierarchy_test PRIVATE assimp)
target_compile_options(scene_hierarchy_test PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
add_executable(material_test src/cpu_mesh.cpp src/cpu_material.cpp src/cpu_scene.cpp src/material_test.cpp)
target_compile_features(material_test PRIVATE cxx_std_17)
target_link_libraries(material_test PRIVATE assimp)
target_compile_options(material_test PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
add_executable(import_scope_test src/cpu_mesh.cpp src/cpu_material.cpp src/cpu_scene.cpp src/import_scope_test.cpp)
target_compile_features(import_scope_test PRIVATE cxx_std_17)
target_link_libraries(import_scope_test PRIVATE assimp)
target_compile_options(import_scope_test PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
add_executable(gpu_upload_test src/cpu_mesh.cpp src/cpu_material.cpp src/cpu_scene.cpp src/gpu_upload.cpp src/upload_preview.cpp)
target_compile_features(gpu_upload_test PRIVATE cxx_std_17)
target_compile_definitions(gpu_upload_test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(gpu_upload_test PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
target_link_libraries(gpu_upload_test PRIVATE assimp d3d12 dxgi d3dcompiler windowscodecs ole32)
