# D.4：target依赖表达职责。PUBLIC传播编译/链接契约，PRIVATE只供当前实现使用。
# 静态库消除重复编译，但它本身不是动态插件或ABI兼容保证。
function(lesson_target name)
  target_compile_features(${name} PUBLIC cxx_std_17)
  target_compile_options(${name} PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
endfunction()

option(PT10_INTEROP "Build CUDA/OIDN external consumer lesson" OFF)
if(PT10_INTEROP)
  enable_language(CUDA)
  find_package(CUDAToolkit 13.3 REQUIRED)
  set(PT_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../../path_tracing/dxr-path-tracer")
  set(OIDN_ROOT "${PT_PATH}/build/deps/oidn-2.5.0.x64.windows")
  if(NOT EXISTS "${OIDN_ROOT}/include/OpenImageDenoise/oidn.h")
    message(FATAL_ERROR "Run dxr-path-tracer/setup_oidn.ps1 first")
  endif()
  add_library(path_cuda STATIC "${PT_PATH}/path_cuda.cu")
  set_target_properties(path_cuda PROPERTIES CUDA_ARCHITECTURES 120 CUDA_STANDARD 17 CUDA_RUNTIME_LIBRARY Shared)
  target_compile_options(path_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/utf-8>)
  target_link_libraries(path_cuda PUBLIC CUDA::cudart)
  add_executable(path_interop_test "${PT_PATH}/path_interop_test.cpp" "${PT_PATH}/path_interop.cpp" "${PT_PATH}/path_oidn.cpp")
  lesson_target(path_interop_test)
  target_include_directories(path_interop_test PRIVATE "${OIDN_ROOT}/include")
  target_link_libraries(path_interop_test PRIVATE scene_import pbr_render path_cuda "${OIDN_ROOT}/lib/OpenImageDenoise.lib")
  set_target_properties(path_interop_test PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PT_PATH}/build" CUDA_RUNTIME_LIBRARY Shared)
  add_executable(path_final "${PT_PATH}/path_final.cpp" "${PT_PATH}/path_interop.cpp" "${PT_PATH}/path_oidn.cpp")
  lesson_target(path_final)
  target_include_directories(path_final PRIVATE "${OIDN_ROOT}/include")
  target_link_libraries(path_final PRIVATE scene_import pbr_render path_cuda "${OIDN_ROOT}/lib/OpenImageDenoise.lib" user32)
  set_target_properties(path_final PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PT_PATH}/build" CUDA_RUNTIME_LIBRARY Shared)
endif()
